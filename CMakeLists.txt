cmake_minimum_required(VERSION 3.20)
project(parallel_heat_solver LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Debug CACHE STRING "Build type" FORCE)
endif()
add_library(heat_solver src/heat.cpp src/cli.cpp)
target_include_directories(heat_solver PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
add_executable(heat_serial src/main.cpp)
target_link_libraries(heat_serial PRIVATE heat_solver)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(heat_solver PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(heat_serial PRIVATE -Wall -Wextra -Wpedantic)
endif()
find_package(MPI REQUIRED COMPONENTS CXX)
add_library(heat_parallel src/heat_mpi.cpp)
target_link_libraries(heat_parallel PUBLIC heat_solver MPI::MPI_CXX)
add_executable(heat_mpi src/main_mpi.cpp)
target_link_libraries(heat_mpi PRIVATE heat_parallel)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(heat_parallel PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(heat_mpi PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(heat_tests tests/test_heat.cpp)
    target_link_libraries(heat_tests PRIVATE heat_solver)
    foreach(case IN ITEMS initial hand parity validation analytic)
        add_test(NAME heat_${case} COMMAND heat_tests ${case})
        set_tests_properties(heat_${case} PROPERTIES TIMEOUT 30)
    endforeach()
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME heat_cli COMMAND ${Python3_EXECUTABLE}
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_cli.py" $<TARGET_FILE:heat_serial>)
    set_tests_properties(heat_cli PROPERTIES TIMEOUT 30)
    set(mpi_fault_args)
    set(mpi_check_dependencies heat_tests heat_serial heat_mpi)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        add_executable(heat_mpi_failures tests/test_mpi_failures.cpp)
        target_link_libraries(heat_mpi_failures PRIVATE heat_parallel)
        target_link_options(heat_mpi_failures PRIVATE "-Wl,--wrap=MPI_Sendrecv")
        target_compile_options(heat_mpi_failures PRIVATE -Wall -Wextra -Wpedantic)
        list(APPEND mpi_fault_args --fault-solver=$<TARGET_FILE:heat_mpi_failures>)
        list(APPEND mpi_check_dependencies heat_mpi_failures)
    endif()
    set(HEAT_MPI_TEST_MAX_PROCS 4 CACHE STRING "MPI regression resources (1..4); not auto-detected")
    if(NOT HEAT_MPI_TEST_MAX_PROCS MATCHES "^[1-4]$")
        message(FATAL_ERROR "HEAT_MPI_TEST_MAX_PROCS must be 1..4")
    endif()
    set(mpi_test_flags)
    foreach(flag IN LISTS MPIEXEC_PREFLAGS)
        list(APPEND mpi_test_flags "--pre=${flag}")
    endforeach()
    foreach(flag IN LISTS MPIEXEC_POSTFLAGS)
        list(APPEND mpi_test_flags "--post=${flag}")
    endforeach()
    foreach(np RANGE 1 ${HEAT_MPI_TEST_MAX_PROCS})
        add_test(NAME heat_mpi_p${np} COMMAND ${Python3_EXECUTABLE}
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_mpi.py"
            --serial=$<TARGET_FILE:heat_serial> --solver=$<TARGET_FILE:heat_mpi>
            --mpiexec=${MPIEXEC_EXECUTABLE} --npflag=${MPIEXEC_NUMPROC_FLAG}
            --processes=${np} ${mpi_test_flags} ${mpi_fault_args})
        set_tests_properties(heat_mpi_p${np} PROPERTIES TIMEOUT 120 PROCESSORS ${np})
    endforeach()
    add_custom_target(check
        COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/scripts/check.py"
            --test-dir "${CMAKE_CURRENT_BINARY_DIR}" --ctest "${CMAKE_CTEST_COMMAND}"
        DEPENDS ${mpi_check_dependencies}
        USES_TERMINAL
        COMMENT "Run CTest with four-decimal test durations")
endif()
