cmake_minimum_required(VERSION 3.20)
project(parallel_heat_solver LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Debug CACHE STRING "Build type" FORCE)
endif()
add_library(heat_solver src/heat.cpp)
target_include_directories(heat_solver PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
add_executable(heat_serial src/main.cpp)
target_link_libraries(heat_serial PRIVATE heat_solver)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(heat_solver PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(heat_serial PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(heat_tests tests/test_heat.cpp)
    target_link_libraries(heat_tests PRIVATE heat_solver)
    foreach(case IN ITEMS initial hand parity validation analytic)
        add_test(NAME heat_${case} COMMAND heat_tests ${case})
        set_tests_properties(heat_${case} PROPERTIES TIMEOUT 30)
    endforeach()
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME heat_cli COMMAND ${Python3_EXECUTABLE}
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_cli.py" $<TARGET_FILE:heat_serial>)
    set_tests_properties(heat_cli PROPERTIES TIMEOUT 30)
    add_custom_target(check
        COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/scripts/check.py"
            --test-dir "${CMAKE_CURRENT_BINARY_DIR}" --ctest "${CMAKE_CTEST_COMMAND}"
        DEPENDS heat_tests heat_serial
        USES_TERMINAL
        COMMENT "Run CTest with four-decimal test durations")
endif()
